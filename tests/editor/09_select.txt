# 09 select with F4 and with Shift and the arrows; F4 cut, F5 copy, F5 paste, Del in mark mode (the PicoMite's keys)
put t.bas start9.bas
cmd EDIT "t.bas"
# F4, Down, F4: the first line cut
key f4
key down
key f4
status 5 lines
line 0 bbb
# pasted above 'ddd'
key down 2
key f5
line 2 aaa
line 3 ddd
# Shift-Down selects 'ddd' and its line end; F5 copies it; pasted after the last line
key shift-down
key f5
key end
key end
key enter
key f5
# Shift-Right twice selects 'cc' of 'ccc'; Del deletes it
key home
key home
key down
key shift-right 2
key del
line 1 c
# Shift-Left selects backwards: from the end of 'aaa', twice, then Del leaves 'a'
key down
key end
key shift-left 2
key del
line 2 a
# Shift-Up from the start of 'ddd' selects 'a' and its line end; Del joins them
key down
key home
key shift-up
key del
line 2 ddd
status 7 lines
# F4, Right, F4 cuts the 'b' of 'bbb'
key home
key home
key f4
key right
key f4
line 0 bb
key f1
prompt
get t.bas expect9.bas
