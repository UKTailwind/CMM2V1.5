# 08 Shift-Tab, Shift-Del, and Tab with OPTION TAB 2, 3, 4 and 8 (the PicoMite's keys and kilo's Shift keys, Peter 19 Oct)
put t.bas start8.bas
cmd OPTION TAB 2
prompt
cmd EDIT "t.bas"
# Shift-Tab deletes up to one tab stop of spaces at the cursor
key down 3
key shift-tab
line 3   indented four
# Shift-Del at the start of a line deletes its leading spaces; elsewhere it deletes the next character
key down
key shift-del
line 4 indented five
key down
key shift-del
line 5 yz
# Tab to the next stop (2)
key end
key enter
key tab
type a
key f1
prompt
get t.bas expect8a.bas
# Tab 4: Shift-Tab deletes four spaces; Tab after one character inserts three
cmd OPTION TAB 4
prompt
cmd EDIT "t.bas"
key down 7
key shift-tab
key up 2
key end
key enter
type e
key tab
type f
key f1
prompt
# Tab 3, then 8, on new lines after the first (no indent to carry)
cmd OPTION TAB 3
prompt
cmd EDIT "t.bas"
key end
key enter
type b
key tab
type c
key enter
key tab
type d
key f1
prompt
cmd OPTION TAB 8
prompt
cmd EDIT "t.bas"
key end
key enter
type g
key tab
type h
key f1
prompt
cmd OPTION TAB 2
prompt
get t.bas expect8.bas
